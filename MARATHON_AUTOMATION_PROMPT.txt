node scripts/marathon-groom.mjs --should-run --mark-skipped

# exit 3 = pure idle, агент не нужен → exit 0 (это и есть экономия токенов)
if [ "$GROOM_EXIT" -ne 0 ]; then exit 0; fi

node scripts/marathon-reconcile.mjs --apply
node scripts/marathon-groom.mjs --apply        # включая done_on_main дедуп
node scripts/marathon-scan.mjs --sync-marathon

PICK=$(node scripts/marathon-scan.mjs --next)
echo "$PICK"

# РЕШЕНИЕ — ровно два сценария, выбор делает groom, а не ты:
#
# СЦЕНАРИЙ А (reason=ok, pick.id есть) → ТОЛЬКО ЗАДАЧА. Self-improvement ЗАПРЕЩЁН.
#   1. Сделать задачу (код/тесты). pnpm typecheck / tests — gate.
#   2. node scripts/marathon-last-run.mjs --task M-NN --result "..."
#   3. Статус M-NN → done в MARATHON.md + запись в TESTLOG.md.
#      Если cat=R (UX-задача) — ДОПОЛНИТЕЛЬНО U-NN: todo → done в UX_BACKLOG.md,
#      иначе groom вернёт задачу в очередь (done_but_active). В TESTLOG указать
#      «было N шагов → стало M» или что именно перестало требовать ручных действий.
#   4. ОДИН коммит: git add -A && git commit -m "feat(marathon): M-NN ..."
#   5. git checkout main && git pull origin main && git merge <run-branch> --no-edit && git push origin main
#
# СЦЕНАРИЙ Б (reason=scanner_empty_expand) → ТОЛЬКО ПОПОЛНИТЬ ОЧЕРЕДЬ. Не «подумать», а сделать.

#     1. Открыть MARATHON.md § Генератор UX-задач → выполнить промпт оттуда
#     2. Вставить новые строки U-NN в UX_BACKLOG.md (status todo)
#     3. node scripts/marathon-scan.mjs --sync-marathon
#     4. ОДИН коммит: "feat(marathon): пополнить UX_BACKLOG (N задач)" → merge main → push
#   Приоритет Б2 — только если UX_BACKLOG тоже пуст: новая техническая категория
#     в scripts/marathon-scan.mjs (см. список категорий в шапке файла) → sync → коммит → main.
